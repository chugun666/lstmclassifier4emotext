class LSTMClassifier(nn.Module):
    def __init__(self, vocab_size, embed_dim, hidden_dim, output_dim):
        super().__init__()
        self.embedding = nn.Embedding(vocab_size, embed_dim, padding_idx=0)
        # num_layers=2 добавляет второй слой LSTM
        self.lstm = nn.LSTM(embed_dim, hidden_dim, num_layers=1, batch_first=True, dropout=0.2)
        self.dropout = nn.Dropout(0.2)
        self.fc = nn.Linear(hidden_dim, output_dim)
        
    def forward(self, text):
        embedded = self.embedding(text)
        # h_n имеет размер [num_layers, batch, hidden_dim]
        _, (hidden, _) = self.lstm(embedded)
        # Берем скрытое состояние последнего (второго) слоя
        out = self.dropout(hidden[-1])
        return self.fc(out)
